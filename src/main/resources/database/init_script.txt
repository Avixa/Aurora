-- =============================================================================
--  ALTN72 - Projet final 2026-27 - Supervision d'équipements et suivi d'incidents
--  Schéma PostgreSQL (version 12 ou supérieure)
--
--  Exécution manuelle  : psql -U <utilisateur> -d <base> -f schema.sql -f data.sql
--  Exécution au démarrage de Spring Boot : placer schema.sql et data.sql dans
--  src/main/resources et définir spring.sql.init.mode=always
--
--  Conventions
--   - Noms en snake_case alignés sur la stratégie de nommage par défaut de
--     Spring Boot (dateOuverture -> date_ouverture, station -> station_id) :
--     les entités n'ont presque pas besoin de @Column ni de @JoinColumn.
--   - Clés primaires BIGINT IDENTITY      -> @GeneratedValue(strategy = GenerationType.IDENTITY)
--   - Énumérations en VARCHAR + CHECK      -> @Enumerated(EnumType.STRING)
--   - Dates en TIMESTAMP (sans fuseau)     -> LocalDateTime
--   - Les CHECK sont un filet de sécurité. Les règles de gestion (§5.8) restent
--     implémentées et testées dans la couche service.
--   - L'indicateur vert / orange / rouge n'est volontairement PAS stocké (§5.5) :
--     il est calculé à partir de date_ouverture et de l'heure courante.
-- =============================================================================

DROP TABLE IF EXISTS
    evenement_incident,
    evaluation,
    incident,
    prestataire_type_equipement,
    equipement,
    prestataire,
    type_equipement,
    station,
    utilisateur
CASCADE;


-- =============================================================================
--  SOCLE COMMUN (§4)
-- =============================================================================

CREATE TABLE utilisateur (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    login         VARCHAR(50)  NOT NULL,
    -- Empreinte BCrypt (60 caractères), jamais le mot de passe en clair (§7.7)
    mot_de_passe  VARCHAR(100) NOT NULL,
    prenom        VARCHAR(50)  NOT NULL,
    profil        VARCHAR(30)  NOT NULL,
    CONSTRAINT uk_utilisateur_login UNIQUE (login),
    CONSTRAINT ck_utilisateur_profil CHECK (profil IN ('RESPONSABLE_EXPLOITATION', 'AGENT_STATION'))
);


CREATE TABLE station (
    id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nom        VARCHAR(100) NOT NULL,
    -- Coordonnées facultatives, prévues pour la piste 4 (cartographie du réseau)
    latitude   DOUBLE PRECISION,
    longitude  DOUBLE PRECISION,
    CONSTRAINT uk_station_nom UNIQUE (nom),
    CONSTRAINT ck_station_latitude    CHECK (latitude  BETWEEN -90  AND 90),
    CONSTRAINT ck_station_longitude   CHECK (longitude BETWEEN -180 AND 180),
    CONSTRAINT ck_station_coordonnees CHECK ((latitude IS NULL) = (longitude IS NULL))
);


-- Table plutôt qu'énumération Java : un nouveau type s'ajoute sans recompiler,
-- et la table sert de pivot aux compétences des prestataires (piste 3) ainsi
-- qu'aux statistiques par type d'équipement (piste 2).
CREATE TABLE type_equipement (
    id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    code     VARCHAR(30)  NOT NULL,
    libelle  VARCHAR(100) NOT NULL,
    CONSTRAINT uk_type_equipement_code    UNIQUE (code),
    CONSTRAINT uk_type_equipement_libelle UNIQUE (libelle)
);


CREATE TABLE equipement (
    id                   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    reference            VARCHAR(30)  NOT NULL,
    libelle              VARCHAR(150) NOT NULL,
    description          VARCHAR(500),
    type_equipement_id   BIGINT       NOT NULL,
    station_id           BIGINT       NOT NULL,
    etat_fonctionnement  VARCHAR(20)  NOT NULL,
    -- RG9 : la référence d'un équipement est unique
    CONSTRAINT uk_equipement_reference UNIQUE (reference),
    CONSTRAINT fk_equipement_type      FOREIGN KEY (type_equipement_id) REFERENCES type_equipement (id),
    CONSTRAINT fk_equipement_station   FOREIGN KEY (station_id)         REFERENCES station (id),
    CONSTRAINT ck_equipement_etat CHECK (etat_fonctionnement IN ('EN_SERVICE', 'DEGRADE', 'HORS_SERVICE'))
);

-- PostgreSQL n'indexe pas automatiquement les clés étrangères
CREATE INDEX idx_equipement_station ON equipement (station_id);          -- F2
CREATE INDEX idx_equipement_type    ON equipement (type_equipement_id);


CREATE TABLE prestataire (
    id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nom        VARCHAR(100) NOT NULL,
    email      VARCHAR(150),
    telephone  VARCHAR(20),
    CONSTRAINT uk_prestataire_nom     UNIQUE (nom),
    -- « De quoi le contacter » : au moins un moyen de contact
    CONSTRAINT ck_prestataire_contact CHECK (email IS NOT NULL OR telephone IS NOT NULL)
);


CREATE TABLE incident (
    id                BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    equipement_id     BIGINT        NOT NULL,
    description       VARCHAR(1000) NOT NULL,
    date_ouverture    TIMESTAMP     NOT NULL,
    statut            VARCHAR(20)   NOT NULL,
    prestataire_id    BIGINT,
    -- Date de la dernière (ré)affectation : sert au délai de prise en charge (pistes 1 et 2)
    date_affectation  TIMESTAMP,
    date_resolution   TIMESTAMP,
    -- Utilisateur connecté au moment de la déclaration (F4), renseigné par le service
    declarant_id      BIGINT,
    CONSTRAINT fk_incident_equipement  FOREIGN KEY (equipement_id)  REFERENCES equipement (id),
    CONSTRAINT fk_incident_prestataire FOREIGN KEY (prestataire_id) REFERENCES prestataire (id),
    CONSTRAINT fk_incident_declarant   FOREIGN KEY (declarant_id)   REFERENCES utilisateur (id),

    -- RG2 : non affecté, affecté ou résolu
    CONSTRAINT ck_incident_statut CHECK (statut IN ('NON_AFFECTE', 'AFFECTE', 'RESOLU')),
    -- RG1 / RG2 : un incident non affecté n'a pas de prestataire, un incident affecté en a un
    CONSTRAINT ck_incident_non_affecte CHECK (statut <> 'NON_AFFECTE' OR prestataire_id IS NULL),
    CONSTRAINT ck_incident_affecte     CHECK (statut <> 'AFFECTE'     OR prestataire_id IS NOT NULL),
    CONSTRAINT ck_incident_affectation_coherente CHECK ((prestataire_id IS NULL) = (date_affectation IS NULL)),
    -- RG5 : un incident résolu, et lui seul, possède une date de résolution
    CONSTRAINT ck_incident_resolution_coherente CHECK ((statut = 'RESOLU') = (date_resolution IS NOT NULL)),
    -- RG5 : chronologie (un CHECK dont une valeur est NULL est considéré comme satisfait)
    CONSTRAINT ck_incident_affectation_apres_ouverture  CHECK (date_affectation >= date_ouverture),
    CONSTRAINT ck_incident_resolution_apres_ouverture   CHECK (date_resolution  >= date_ouverture),
    CONSTRAINT ck_incident_resolution_apres_affectation CHECK (date_resolution  >= date_affectation)
);

CREATE INDEX idx_incident_equipement_ouverture ON incident (equipement_id, date_ouverture DESC);  -- F3, RG7
CREATE INDEX idx_incident_actifs     ON incident (date_ouverture)       WHERE statut <> 'RESOLU'; -- F11, RG6
CREATE INDEX idx_incident_historique ON incident (date_resolution DESC) WHERE statut = 'RESOLU';  -- F9, RG7
CREATE INDEX idx_incident_prestataire ON incident (prestataire_id);
CREATE INDEX idx_incident_declarant   ON incident (declarant_id);


-- RG8 : la condition « incident résolu ET affecté à un prestataire » porte sur
-- deux tables. Elle est vérifiée dans le service, pas par une contrainte.
CREATE TABLE evaluation (
    id               BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    incident_id      BIGINT        NOT NULL,
    note             INTEGER       NOT NULL,
    commentaire      VARCHAR(1000),
    date_evaluation  TIMESTAMP     NOT NULL,
    CONSTRAINT uk_evaluation_incident UNIQUE (incident_id),            -- RG8 : une seule par incident
    CONSTRAINT fk_evaluation_incident FOREIGN KEY (incident_id) REFERENCES incident (id),
    CONSTRAINT ck_evaluation_note     CHECK (note BETWEEN 1 AND 5)     -- RG8 : entier de 1 à 5
);


-- =============================================================================
--  ANTICIPATION DU LOT ALPHA (§6)
--  Tables autonomes : à supprimer si la piste correspondante n'est pas retenue.
-- =============================================================================

-- Piste 3 (aide à l'affectation) : types d'équipements traités par chaque prestataire
CREATE TABLE prestataire_type_equipement (
    prestataire_id      BIGINT NOT NULL,
    type_equipement_id  BIGINT NOT NULL,
    CONSTRAINT pk_prestataire_type_equipement PRIMARY KEY (prestataire_id, type_equipement_id),
    CONSTRAINT fk_competence_prestataire FOREIGN KEY (prestataire_id)     REFERENCES prestataire (id)     ON DELETE CASCADE,
    CONSTRAINT fk_competence_type        FOREIGN KEY (type_equipement_id) REFERENCES type_equipement (id) ON DELETE CASCADE
);

CREATE INDEX idx_competence_type ON prestataire_type_equipement (type_equipement_id);


-- Piste 5 (journal métier) : qui a fait quoi, et quand, sur un incident
CREATE TABLE evenement_incident (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    incident_id     BIGINT       NOT NULL,
    auteur_id       BIGINT,
    date_evenement  TIMESTAMP    NOT NULL,
    type_evenement  VARCHAR(30)  NOT NULL,
    detail          VARCHAR(500),
    CONSTRAINT fk_evenement_incident FOREIGN KEY (incident_id) REFERENCES incident (id) ON DELETE CASCADE,
    CONSTRAINT fk_evenement_auteur   FOREIGN KEY (auteur_id)   REFERENCES utilisateur (id),
    CONSTRAINT ck_evenement_type CHECK (type_evenement IN
        ('DECLARATION', 'MODIFICATION', 'AFFECTATION', 'REAFFECTATION', 'RESOLUTION', 'EVALUATION'))
);

CREATE INDEX idx_evenement_incident_date ON evenement_incident (incident_id, date_evenement);
